Inspect approved SQL and bind values ⌄

The application selects this fixed SQL for TURNOVER_RATE. The question supplies only approved bind values. The end date is exclusive.

WITH monthly_hc AS (
  SELECT h.month_start, SUM(h.headcount) AS monthly_count
  FROM WH_HEADCOUNT_MONTH h
  JOIN WH_LOCATION l ON l.location_id = h.location_id
  WHERE h.month_start >= :start_date
    AND h.month_start < :end_date
    AND l.state = :state
  GROUP BY h.month_start
), denom AS (
  SELECT AVG(monthly_count) AS avg_headcount
  FROM monthly_hc
  HAVING COUNT(*) = :expected_months
), numer AS (
  SELECT COUNT(*) AS terminations
  FROM WH_TERMINATION t
  JOIN WH_LOCATION l ON l.location_id = t.location_id
  WHERE t.term_date >= :start_date
    AND t.term_date < :end_date
    AND l.state = :state
    AND t.term_type <> 'INTERNAL_TRANSFER'
)
SELECT ROUND(100 * NVL(n.terminations, 0) / d.avg_headcount, 4) AS value
FROM denom d LEFT JOIN numer n ON 1 = 1
WHERE d.avg_headcount > 0
